target_compile_options(trace_core PRIVATE /W4 /EHsc /utf-8 /fp:precise)

target_compile_options(trace_engine PRIVATE /W4 /EHsc /utf-8 /fp:precise)
target_link_libraries(trace_engine PRIVATE trace_core "${ELECTRON_SDK}/x64/node.lib")
set_target_properties(trace_engine PROPERTIES PREFIX "" SUFFIX ".node")
